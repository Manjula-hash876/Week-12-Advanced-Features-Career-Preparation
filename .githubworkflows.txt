# 1. Frontend CI Workflow

## .github/workflows/frontend-ci.yml

```yaml id="hks8fr"
name: Frontend Continuous Integration

on:

  push:
    branches:
      - main
      - develop

  pull_request:
    branches:
      - main

jobs:

  frontend-build:

    runs-on: ubuntu-latest

    defaults:
      run:
        working-directory: frontend

    steps:

      - name: Checkout Repository
        uses: actions/checkout@v3

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: 18

      - name: Install Frontend Dependencies
        run: npm install

      - name: Run Frontend Linting
        run: npm run lint

      - name: Run Frontend Tests
        run: npm test

      - name: Build Frontend Application
        run: npm run build

      - name: Upload Frontend Build Artifact
        uses: actions/upload-artifact@v3
        with:
          name: frontend-build
          path: frontend/dist
```

# 2. Backend CI Workflow

## .github/workflows/backend-ci.yml

```yaml id="2r2p2o"
name: Backend Continuous Integration

on:

  push:
    branches:
      - main
      - develop

jobs:

  backend-build:

    runs-on: ubuntu-latest

    defaults:
      run:
        working-directory: backend

    services:

      mongodb:
        image: mongo:6

        ports:
          - 27017:27017

    steps:

      - name: Checkout Repository
        uses: actions/checkout@v3

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: 18

      - name: Install Backend Dependencies
        run: npm install

      - name: Create Environment Variables
        run: |
          echo "PORT=5000" >> .env
          echo "MONGO_DB_URL=mongodb://localhost:27017/socialmedia" >> .env
          echo "JWT_SECRET_KEY=advancedSecretKey" >> .env

      - name: Run Backend Tests
        run: npm test

      - name: Build Backend Application
        run: echo "Backend Build Successful"
```

# 4. Docker Deployment Workflow

## .github/workflows/docker-deploy.yml

```yaml id="0nd7t7"
name: Docker Deployment Pipeline

on:

  push:
    branches:
      - main

jobs:

  docker-deployment:

    runs-on: ubuntu-latest

    steps:

      - name: Checkout Source Code
        uses: actions/checkout@v3

      - name: Login to DockerHub
        uses: docker/login-action@v2
        with:
          username: ${{ secrets.DOCKER_USERNAME }}
          password: ${{ secrets.DOCKER_PASSWORD }}

      - name: Build Frontend Docker Image
        run: |
          docker build \
          -t advanced-socialmedia-frontend \
          ./frontend

      - name: Build Backend Docker Image
        run: |
          docker build \
          -t advanced-socialmedia-backend \
          ./backend

      - name: Start Docker Compose Services
        run: |
          docker-compose \
          -f docker/docker-compose.yml \
          up -d
```

# 5. Security Scan Workflow

## .github/workflows/security-scan.yml

```yaml id="ah0x08"
name: Security Scan Workflow

on:

  push:
    branches:
      - main

jobs:

  dependency-scan:

    runs-on: ubuntu-latest

    steps:

      - name: Checkout Repository
        uses: actions/checkout@v3

      - name: Run npm Audit
        run: |
          cd backend && npm audit --audit-level=high
          cd ../frontend && npm audit --audit-level=high

      - name: Run GitHub CodeQL Analysis
        uses: github/codeql-action/init@v2
        with:
          languages: javascript

      - name: Autobuild
        uses: github/codeql-action/autobuild@v2

      - name: Perform CodeQL Analysis
        uses: github/codeql-action/analyze@v2
```

# 7. Production Deployment Workflow

## .github/workflows/production-deploy.yml

```yaml id="jlwm7q"
name: Production Deployment Workflow

on:

  workflow_run:
    workflows:
      - Frontend Continuous Integration
      - Backend Continuous Integration

    types:
      - completed

jobs:

  production-deploy:

    runs-on: ubuntu-latest

    if: ${{ github.event.workflow_run.conclusion == 'success' }}

    steps:

      - name: Checkout Repository
        uses: actions/checkout@v3

      - name: Setup Node.js Environment
        uses: actions/setup-node@v3
        with:
          node-version: 18

      - name: Install Deployment CLI
        run: npm install -g vercel

      - name: Deploy Frontend to Vercel
        run: |
          vercel deploy \
          --prod \
          --token=${{ secrets.VERCEL_TOKEN }}

      - name: Deploy Backend to Render
        run: echo "Backend Deployment Triggered"
```
