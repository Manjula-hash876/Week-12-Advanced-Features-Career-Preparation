# 1. Axios Configuration

## src/api/axiosConfig.js

```javascript
import axios from "axios";

const API = axios.create({
  baseURL: import.meta.env.VITE_API_URL,
  withCredentials: true,
});

API.interceptors.request.use((config) => {
  const authToken = localStorage.getItem("authToken");

  if (authToken) {
    config.headers.Authorization = `Bearer ${authToken}`;
  }

  return config;
});

export default API;
```

# 2. Authentication Context

## src/context/AuthContext.jsx

```javascript
import { createContext, useContext, useState } from "react";

const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  const [currentUser, setCurrentUser] = useState(null);

  const loginUser = (userData, token) => {
    localStorage.setItem("authToken", token);
    setCurrentUser(userData);
  };

  const logoutUser = () => {
    localStorage.removeItem("authToken");
    setCurrentUser(null);
  };

  return (
    <AuthContext.Provider
      value={{
        currentUser,
        loginUser,
        logoutUser,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);
```

# 3. Socket.io Hook

## src/hooks/useSocket.js

```javascript
import { useEffect, useState } from "react";
import io from "socket.io-client";
import { useAuth } from "../context/AuthContext";

const SOCKET_SERVER_URL = import.meta.env.VITE_API_URL;

export const useSocket = () => {
  const { currentUser } = useAuth();
  const [socketClient, setSocketClient] = useState(null);

  useEffect(() => {
    if (!currentUser) return;

    const newSocket = io(SOCKET_SERVER_URL, {
      auth: {
        token: localStorage.getItem("authToken"),
      },
    });

    newSocket.on("connect", () => {
      console.log("Socket Connected:", newSocket.id);

      newSocket.emit("join-user-room", currentUser._id);
    });

    setSocketClient(newSocket);

    return () => {
      newSocket.disconnect();
    };
  }, [currentUser]);

  return socketClient;
};
```

# 4. Chat Window Component

## src/components/chat/ChatWindow.jsx

```javascript
import { useEffect, useState } from "react";
import { useSocket } from "../../hooks/useSocket";

const ChatWindow = ({ selectedChatRoom }) => {
  const socketClient = useSocket();

  const [chatMessages, setChatMessages] = useState([]);
  const [typingStatus, setTypingStatus] = useState("");

  useEffect(() => {
    if (!socketClient || !selectedChatRoom) return;

    socketClient.emit("join-chat-room", selectedChatRoom);

    socketClient.on("receive-message", (incomingMessage) => {
      setChatMessages((prevMessages) => [
        ...prevMessages,
        incomingMessage,
      ]);
    });

    socketClient.on("user-typing", (data) => {
      setTypingStatus(data.message);
    });

    return () => {
      socketClient.off("receive-message");
    };
  }, [socketClient, selectedChatRoom]);

  return (
    <div className="chat-window">
      <h2>Chat Room</h2>

      {chatMessages.map((message, index) => (
        <div key={index}>
          <strong>{message.senderName}</strong>
          <p>{message.content}</p>
        </div>
      ))}

      <p>{typingStatus}</p>
    </div>
  );
};

export default ChatWindow;
```

# 5. Media Upload Component

## src/components/upload/MediaUpload.jsx

```javascript
import { useState } from "react";
import axios from "axios";

const MediaUpload = () => {
  const [selectedFile, setSelectedFile] = useState(null);
  const [uploadStatus, setUploadStatus] = useState("");

  const handleFileChange = (event) => {
    setSelectedFile(event.target.files[0]);
  };

  const uploadMediaFile = async () => {
    if (!selectedFile) return;

    const formData = new FormData();
    formData.append("mediaFile", selectedFile);

    try {
      setUploadStatus("Uploading...");

      const response = await axios.post(
        `${import.meta.env.VITE_API_URL}/upload`,
        formData
      );

      setUploadStatus("Upload Successful");
      console.log(response.data);
    } catch (error) {
      setUploadStatus("Upload Failed");
      console.error(error);
    }
  };

  return (
    <div>
      <input type="file" onChange={handleFileChange} />

      <button onClick={uploadMediaFile}>
        Upload File
      </button>

      <p>{uploadStatus}</p>
    </div>
  );
};

export default MediaUpload;
```

# 6. Notification Bell Component

## src/components/notifications/NotificationBell.jsx

```javascript
import { useEffect, useState } from "react";
import { useSocket } from "../../hooks/useSocket";

const NotificationBell = () => {
  const socketClient = useSocket();

  const [notifications, setNotifications] = useState([]);

  useEffect(() => {
    if (!socketClient) return;

    socketClient.on("new-notification", (notificationData) => {
      setNotifications((prev) => [
        notificationData,
        ...prev,
      ]);
    });

    return () => {
      socketClient.off("new-notification");
    };
  }, [socketClient]);

  return (
    <div>
      <button>
        Notifications ({notifications.length})
      </button>

      <div>
        {notifications.map((notification, index) => (
          <p key={index}>
            {notification.message}
          </p>
        ))}
      </div>
    </div>
  );
};

export default NotificationBell;
```

# 7. Protected Route

## src/components/common/ProtectedRoute.jsx

```javascript
import { Navigate } from "react-router-dom";

const ProtectedRoute = ({ children }) => {
  const authToken = localStorage.getItem("authToken");

  if (!authToken) {
    return <Navigate to="/login" />;
  }

  return children;
};

export default ProtectedRoute;
```
