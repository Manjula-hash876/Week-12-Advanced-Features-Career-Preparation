# 3. Frontend Login Page Test

## tests/frontend/LoginPage.test.jsx

```javascript
import { render, screen, fireEvent } from "@testing-library/react";

import LoginPage from "../../src/pages/LoginPage";

describe("Login Page Component", () => {

  test("renders login form correctly", () => {

    render(<LoginPage />);

    expect(
      screen.getByText(/login/i)
    ).toBeInTheDocument();

    expect(
      screen.getByPlaceholderText(/email/i)
    ).toBeInTheDocument();

    expect(
      screen.getByPlaceholderText(/password/i)
    ).toBeInTheDocument();
  });

  test("updates input fields correctly", () => {

    render(<LoginPage />);

    const emailInputField =
      screen.getByPlaceholderText(/email/i);

    const passwordInputField =
      screen.getByPlaceholderText(/password/i);

    fireEvent.change(emailInputField, {
      target: {
        value: "test@example.com",
      },
    });

    fireEvent.change(passwordInputField, {
      target: {
        value: "Password123",
      },
    });

    expect(emailInputField.value).toBe(
      "test@example.com"
    );

    expect(passwordInputField.value).toBe(
      "Password123"
    );
  });

});
```

# 6. Media Upload Component Test

## tests/frontend/MediaUpload.test.jsx

```javascript
import { render, screen, fireEvent } from "@testing-library/react";

import MediaUpload from
"../../src/components/upload/MediaUpload";

describe("Media Upload Component", () => {

  test("renders upload button", () => {

    render(<MediaUpload />);

    expect(
      screen.getByText(/upload file/i)
    ).toBeInTheDocument();
  });

  test("handles file selection", () => {

    render(<MediaUpload />);

    const fileInputField =
      screen.getByRole("textbox");

    const sampleFile = new File(
      ["sample"],
      "sample.png",
      {
        type: "image/png",
      }
    );

    fireEvent.change(fileInputField, {
      target: {
        files: [sampleFile],
      },
    });

    expect(fileInputField.files[0]).toBe(sampleFile);
  });

});
```

# 7. Backend Authentication API Test

## tests/backend/auth.test.js

```javascript
const request = require("supertest");

const app = require("../../backend/app");

describe("Authentication API", () => {

  test("registers a new user successfully", async () => {

    const userPayload = {
      fullName: "John Smith",
      emailAddress: "johnsmith@example.com",
      password: "Password123",
    };

    const response = await request(app)
      .post("/api/auth/register")
      .send(userPayload);

    expect(response.statusCode).toBe(201);

    expect(response.body.success).toBe(true);

    expect(response.body.user.emailAddress)
      .toBe(userPayload.emailAddress);
  });

});
```

# 11. Socket.io Event Test

## tests/backend/socket.test.js

```javascript
const io = require("socket.io-client");

describe("Socket.io Messaging", () => {

  let clientSocket;

  beforeAll((done) => {

    clientSocket = io("http://localhost:5000");

    clientSocket.on("connect", done);
  });

  afterAll(() => {
    clientSocket.disconnect();
  });

  test("sends and receives messages", (done) => {

    const testMessage = {
      chatRoomId: "room-101",
      senderId: "user-1",
      receiverId: "user-2",
      messageContent: "Hello Testing",
    };

    clientSocket.emit("send-message", testMessage);

    clientSocket.on(
      "receive-message",
      (receivedMessage) => {

        expect(receivedMessage.messageContent)
          .toBe("Hello Testing");

        done();
      }
    );
  });

});
```

# 13. Security Testing Example

## tests/backend/security.test.js

```javascript
const request = require("supertest");

const app = require("../../backend/app");

describe("Security Validation", () => {

  test("blocks unauthorized access", async () => {

    const response = await request(app)
      .get("/api/chat/private");

    expect(response.statusCode).toBe(401);
  });

  test("prevents NoSQL injection", async () => {

    const maliciousPayload = {
      emailAddress: {
        "$gt": ""
      },
      password: {
        "$gt": ""
      }
    };

    const response = await request(app)
      .post("/api/auth/login")
      .send(maliciousPayload);

    expect(response.statusCode).toBe(400);
  });

});
```

# 16. CI/CD Testing Workflow

## .github/workflows/test.yml

```yaml
name: Full Stack Testing Pipeline

on:
  push:
    branches:
      - main

jobs:

  frontend-tests:

    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3

      - name: Install Frontend Dependencies
        run: cd frontend && npm install

      - name: Run Frontend Tests
        run: cd frontend && npm test

  backend-tests:

    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3

      - name: Install Backend Dependencies
        run: cd backend && npm install

      - name: Run Backend Tests
        run: cd backend && npm test
```
